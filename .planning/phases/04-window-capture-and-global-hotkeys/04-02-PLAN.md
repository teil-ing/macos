At file scope (outside any type), define the KeyboardShortcuts.Name extensions with the user's locked default shortcuts:

```swift
import AppKit
import KeyboardShortcuts

extension KeyboardShortcuts.Name {
    static let regionCapture = Self(
        "regionCapture",
        default: .init(.x, modifiers: [.command, .shift])
    )
    static let fullscreenCapture = Self(
        "fullscreenCapture",
        default: .init(.s, modifiers: [.command, .shift])
    )
    static let windowCapture = Self(
        "windowCapture",
        default: .init(.c, modifiers: [.command, .shift])
    )
}
```

These MUST be at file scope (not nested in a type) for Swift 6 strict concurrency compliance. The `default:` parameter ensures the shortcut is only written to UserDefaults if no entry exists yet — user changes in Phase 8 will persist.

The HotkeyMonitor class:

```swift
@MainActor
final class HotkeyMonitor {

    private var wakeObserver: NSObjectProtocol?

    init() {}

    func start(
        onRegion: @escaping @MainActor () -> Void,
        onFullscreen: @escaping @MainActor () -> Void,
        onWindow: @escaping @MainActor () -> Void
    ) {
        // Register handlers EXACTLY ONCE — onKeyUp appends to an array,
        // calling it again would create duplicate handlers (Pitfall 2).
        KeyboardShortcuts.onKeyUp(for: .regionCapture) {
            Task { @MainActor in onRegion() }
        }
        KeyboardShortcuts.onKeyUp(for: .fullscreenCapture) {
            Task { @MainActor in onFullscreen() }
        }
        KeyboardShortcuts.onKeyUp(for: .windowCapture) {
            Task { @MainActor in onWindow() }
        }

        // Re-register after sleep — Carbon hotkeys can fail to re-activate.
        // Only disable/enable — do NOT re-call onKeyUp (would duplicate handlers).
        wakeObserver = NSWorkspace.shared.notificationCenter.addObserver(
            forName: NSWorkspace.didWakeNotification,
            object: nil,
            queue: .main
        ) { _ in
            Task { @MainActor in
                KeyboardShortcuts.disable(.regionCapture, .fullscreenCapture, .windowCapture)
                KeyboardShortcuts.enable(.regionCapture, .fullscreenCapture, .windowCapture)
            }
        }
    }

    func stop() {
        if let observer = wakeObserver {
            NSWorkspace.shared.notificationCenter.removeObserver(observer)
            wakeObserver = nil
        }
        KeyboardShortcuts.disable(.regionCapture, .fullscreenCapture, .windowCapture)
    }
}
```

Key implementation notes:
- `@MainActor` class (NOT actor) — Carbon event registration requires the main run loop.
- `onKeyUp` is used (not `onKeyDown`) — fires after the user lifts the key, preventing the key-up event from being consumed by the overlay.
- The sleep/wake observer only calls `disable`/`enable` — it does NOT re-register handlers. This prevents the handler accumulation bug (research Pitfall 2).
- Conflict handling is "silent" per user decision — if another app holds the shortcut, teil.ing simply doesn't get it. KeyboardShortcuts handles this automatically.

2. Add a `setupHotkeyMonitor()` method:
```swift
private func setupHotkeyMonitor() {
    hotkeyMonitor.start(
        onRegion: { [weak self] in
            Task { @MainActor [weak self] in
                // 200ms delay per locked decision — lets user lift fingers off keys
                try? await Task.sleep(for: .milliseconds(200))
                // Fullscreen hotkey captures the display where the cursor currently is
                // (handled by captureEngine.captureFullscreen which already uses mouse location)
                // Region and window modes show overlays that don't steal focus
                self?.startRegionCaptureFromHotkey()
            }
        },
        onFullscreen: { [weak self] in
            Task { @MainActor [weak self] in
                try? await Task.sleep(for: .milliseconds(200))
                self?.startFullscreenCaptureFromHotkey()
            }
        },
        onWindow: { [weak self] in
            Task { @MainActor [weak self] in
                try? await Task.sleep(for: .milliseconds(200))
                self?.startWindowCaptureFromHotkey()
            }
        }
    )
}
```

3. Add hotkey-specific capture methods. These are similar to the menu-bar-triggered methods but WITHOUT the `closePopover()` call and WITHOUT the initial 200ms delay (the delay is in the hotkey handler above). They reuse the same core capture logic:

```swift
// MARK: - Hotkey-Triggered Capture

private func startRegionCaptureFromHotkey() {
    Task { @MainActor in
        guard let selectedRect = await overlayCoordinator.beginRegionSelection() else { return }
        try? await Task.sleep(for: .milliseconds(50))
        do {
            let result = try await captureEngine.captureRegion(selectedRect)
            lastCaptureResult = result
            await CaptureFeedback.showCaptureFlash(in: selectedRect)
            CaptureFeedback.playCaptureSound()
            CaptureFeedback.showSuccessIcon(on: statusItem)
        } catch {
            showCaptureError(error.localizedDescription)
        }
    }
}

private func startFullscreenCaptureFromHotkey() {
    Task { @MainActor in
        do {
            let result = try await captureEngine.captureFullscreen()
            lastCaptureResult = result
            await CaptureFeedback.showCaptureFlash(in: result.capturedRect)
            CaptureFeedback.playCaptureSound()
            CaptureFeedback.showSuccessIcon(on: statusItem)
        } catch {
            showCaptureError(error.localizedDescription)
        }
    }
}

private func startWindowCaptureFromHotkey() {
    Task { @MainActor in
        guard let selection = await windowSelectionCoordinator.beginWindowSelection() else { return }
        try? await Task.sleep(for: .milliseconds(50))
        switch selection {
        case .window(let scWindow):
            do {
                let result = try await captureEngine.captureWindow(scWindow)
                lastCaptureResult = result
                let appKitRect = windowSelectionCoordinator.cgFrameToAppKit(scWindow.frame)
                await CaptureFeedback.showCaptureFlash(in: appKitRect)
                CaptureFeedback.playCaptureSound()
                CaptureFeedback.showSuccessIcon(on: statusItem)
            } catch {
                showCaptureError(error.localizedDescription)
            }
        case .desktop:
            do {
                let result = try await captureEngine.captureFullscreen()
                lastCaptureResult = result
                await CaptureFeedback.showCaptureFlash(in: result.capturedRect)
                CaptureFeedback.playCaptureSound()
                CaptureFeedback.showSuccessIcon(on: statusItem)
            } catch {
                showCaptureError(error.localizedDescription)
            }
        }
    }
}
```

Note: An alternative to duplicating capture logic is to refactor startRegionCapture/startFullscreenCapture/startWindowCapture to accept a `fromHotkey: Bool` parameter (skip popover close when true, skip initial delay when true). Use your judgment — if refactoring is clean, do it. If it makes the code harder to follow, keep separate methods. The key constraint is: hotkey path must NOT call closePopover() and must NOT have a double 200ms delay. The 200ms is applied once in setupHotkeyMonitor's closures.
